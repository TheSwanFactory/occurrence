"""Fresh stdlib-only anonymous-incidence reconstruction; no repository imports."""
import collections
import hashlib
import itertools
import json
import math
from pathlib import Path

ROOT = Path(__file__).resolve().parent


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":")).encode()


def observe(events):
    # Values are opaque equality tokens. Never parse habitat text or use axis
    # magnitude, coefficient, P, S, delta, block membership, or targets here.
    habitats, left, right = {}, {}, {}
    edges = []
    for event in events:
        axes = [term.split(":")[0] for term in event["ray"].split(",")]
        assert len(axes) == 2
        h = habitats.setdefault(event["habitat"], len(habitats))
        l = left.setdefault(axes[0], len(left))
        r = right.setdefault(axes[1], len(right))
        edges.append((h, l, r))
    return edges, (len(habitats), len(left), len(right))


def reconstruct(edges, sizes):
    nh, nl, nr = sizes
    supports = [collections.Counter((l, r) for h, l, r in edges if h == i)
                for i in range(nh)]
    omissions = [(sorted(set(range(nl)) - {l for l, r in s}),
                  sorted(set(range(nr)) - {r for l, r in s})) for s in supports]
    assert all(len(l) == len(r) == 1 for l, r in omissions)
    l_to_r = {}
    for (l,), (r,) in omissions:
        assert l not in l_to_r or l_to_r[l] == r
        l_to_r[l] = r
    bijection = len(l_to_r) == nl == nr == len(set(l_to_r.values()))
    assert bijection
    r_to_l = {r: l for l, r in l_to_r.items()}

    def signature(counter):
        return tuple(sorted((l, r, n) for (l, r), n in counter.items()))

    grouped = collections.defaultdict(list)
    for h, s in enumerate(supports):
        grouped[signature(s)].append(h)
    partition = sorted(grouped.values())
    reciprocal = []
    lines = set()
    for h, s in enumerate(supports):
        transported = {(l, r_to_l[r]) for l, r in s}
        assert all(a != b and (b, a) in transported for a, b in transported)
        pairs = sorted({tuple(sorted((a, b))) for a, b in transported})
        assert collections.Counter(v for p in pairs for v in p) == collections.Counter(
            set(range(nl)) - set(omissions[h][0]))
        reciprocal.append([list(p) for p in pairs])
        for a, b in pairs:
            lines.add(tuple(sorted((omissions[h][0][0], a, b))))
    pair_counts = collections.Counter(p for line in lines for p in itertools.combinations(line, 2))
    pair_unique = set(pair_counts) == set(itertools.combinations(range(nl), 2)) and all(
        count == 1 for count in pair_counts.values())
    plane_autos = 0
    graph_lifts = 0
    graph_base_permutations = 0
    target_signatures = collections.Counter(signature(s) for s in supports)
    # Exhaust ALL L permutations. Any typed graph automorphism must preserve
    # omissions, so its R permutation is forced by the recovered bijection.
    # Count all possible habitat matches and all duplicate-Event permutations.
    for p in itertools.permutations(range(nl)):
        transformed_lines = {tuple(sorted(p[x] for x in line)) for line in lines}
        plane_autos += transformed_lines == lines
        q = {r: l_to_r[p[r_to_l[r]]] for r in range(nr)}
        transformed = [collections.Counter({(p[l], q[r]): n for (l, r), n in s.items()})
                       for s in supports]
        if collections.Counter(signature(s) for s in transformed) == target_signatures:
            graph_base_permutations += 1
            habitat_choices = math.prod(math.factorial(n) for n in target_signatures.values())
            event_choices = math.prod(math.factorial(n) for s in supports for n in s.values())
            graph_lifts += habitat_choices * event_choices
    return {
        "node_counts": {"H": nh, "E": len(edges), "L": nl, "R": nr},
        "omitted_counts_by_habitat": [[len(l), len(r)] for l, r in omissions],
        "omitted_classes_by_habitat": [[l[0], r[0]] for l, r in omissions],
        "identical_unsigned_incidence_partition": partition,
        "omitted_pairs_induce_bijection": bijection,
        "reciprocal_pairs_by_habitat": reciprocal,
        "deduplicated_lines": [list(line) for line in sorted(lines)],
        "deduplicated_line_count": len(lines),
        "every_point_pair_on_exactly_one_line": pair_unique,
        "plane_automorphism_order": plane_autos,
        "typed_observed_graph_automorphism_order": graph_lifts,
        "graph_induced_L_permutations": graph_base_permutations,
    }


def certify(events, edges, reconstructed):
    # ONLY NOW read hidden P. Map each missing anonymous point to the certified
    # habitat P; compare reconstructed triples to the nonzero F_2^3 lines.
    hidden = {}
    for event, (h, _, _) in zip(events, edges):
        point = reconstructed["omitted_classes_by_habitat"][h][0]
        assert point not in hidden or hidden[point] == event["P"]
        hidden[point] = event["P"]
    certified_lines = {p for p in itertools.combinations(range(1, 8), 3)
                       if p[0] ^ p[1] ^ p[2] == 0}
    image = {tuple(sorted(hidden[p] for p in line))
             for line in reconstructed["deduplicated_lines"]}
    return len(set(hidden.values())) == 7 and image == certified_lines


if __name__ == "__main__":
    dataset = json.loads((ROOT / "task_dataset.json").read_text())
    events = dataset["catalogue"]["events"]
    edges, sizes = observe(events)
    result = reconstruct(edges, sizes)
    result["isomorphic_to_certified_hidden_fano_plane"] = certify(events, edges, result)
    # Metadata/coefficient/opaque-name changes must leave the observed object
    # byte-identical; incidence row ordering alone is the naming convention.
    stripped = [{"habitat": "opaque-" + str(edges[i][0]),
                 "ray": f'left-{edges[i][1]}:999,right-{edges[i][2]}:-333'}
                for i in range(len(events))]
    assert observe(stripped) == (edges, sizes)
    result["hidden_metadata_and_numeric_identity_erasure_check"] = True
    output = canonical(result)
    (ROOT / "stage_a.canonical.json").write_bytes(output)
    digest = hashlib.sha256(output).hexdigest()
    (ROOT / "stage_a.sha256").write_text(digest + "\n")
    print(output.decode())
    print("SHA-256", digest, "bytes", len(output))
